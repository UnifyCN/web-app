// @vitest-environment jsdom
// Fresh real-hook and real-editor proof with scripted Auth callbacks.
import { act, cleanup, fireEvent, render, screen, waitFor } from '@testing-library/react';
import { QueryClient, QueryClientProvider } from '@tanstack/react-query';
import { I18nextProvider } from 'react-i18next';
import { useState } from 'react';
import { afterEach, beforeEach, expect, it, vi } from 'vitest';
import { useTaskActionStore, type TaskProgressView } from '@/hooks/useTaskActions';
import { AUTH_USER_KEY } from '@/hooks/useAuthUser';
import { ownAccountWork, noteAccount } from '@/lib/accountSession';
import { clearTaskActionStores } from '@/lib/companionActions/taskActionStore';
import { ReminderSection } from '@/components/companion/tasks/ReminderSection';
import { TaskActionsDrawer } from '@/components/companion/tasks/TaskActionsDrawer';
import { testI18n, scriptedServer } from '@/components/companion/tasks/taskActionsTestHarness';
import en from '@/lib/i18n/locales/en/translation.json';
const auth = vi.hoisted(() => ({
 user: { id: 'real-auth-a' },
 listeners: new Set<(event: string, session: unknown) => void>(),
 reads: 0,
}));
vi.mock('@/lib/supabase/client', () => ({
 isSupabaseConfigured: () => true,
 createClient: () => ({ auth: {
  getSession: async () => { auth.reads++; return { data: { session: { user: {...auth.user} } } }; },
  onAuthStateChange: (f: (event: string, session: unknown) => void) => {
   auth.listeners.add(f);
   return {data:{subscription:{unsubscribe:()=>auth.listeners.delete(f)}}};
  },
 } }),
}));
vi.mock('@/lib/posthog', () => ({ isPostHogConfigured: () => false, posthog: {capture:()=>{}} }));
vi.mock('@/hooks/useChecklist',()=>({TASKS_KEY:['tasks'],useTasks:()=>({data:[{id:'1',title:'Private drawer task',isCustom:true,completed:false,dismissedAt:null}],isLoading:false})}));
const view: TaskProgressView = {
 status:'ready',byRef:new Map(),stepsByRef:new Map(),details:[],stale:false,canWrite:true,
 refetch:()=>{},reminders:{items:[],scheduler:'ready'},
};
let query: QueryClient;
let renders = 0;
function Caller() {
 return <TaskActionsDrawer open onClose={()=>{}} surface="companion"/>;
}
beforeEach(() => {
 const server=scriptedServer({'custom:1':{revision:1}});
 vi.stubGlobal('fetch', async (url: unknown,init: RequestInit)=>{
  const response=await server.fetch(url,init);
  if(JSON.parse(String(init.body)).action==='read'){
   const body=await response.json();
   body.reminders={scheduler:'ready',items:[{
    reminder_id:'r1',task_ref:'custom:1',title:'Private drawer task',
    remind_at:'2027-11-07T05:30:00Z',time_zone:'America/New_York',
    wall_time:'2027-11-07T01:30:00',revision:1,status:'pending',due:true,
    due_at:'2027-11-07T05:30:01Z',cancelled_reason:null,history:[],
   }]};
   return new Response(JSON.stringify(body),{status:response.status,headers:{'content-type':'application/json'}});
  }
  return response;
 });
 clearTaskActionStores(); auth.reads=0; renders=0; noteAccount(auth.user.id);
 query = new QueryClient({ defaultOptions:{queries:{retry:false,gcTime:Infinity}} });
});
afterEach(() => {cleanup(); query.clear(); clearTaskActionStores(); auth.listeners.clear(); vi.unstubAllGlobals();});
async function mount() {
 render(<QueryClientProvider client={query}><I18nextProvider i18n={testI18n()}><Caller/></I18nextProvider></QueryClientProvider>);
 fireEvent.click(await screen.findByRole('button',{name:en.reminders.openTask}));
 await screen.findByLabelText(en.reminders.instant);
 await waitFor(()=>expect(query.isFetching()).toBe(0));
 fireEvent.change(screen.getByLabelText(en.reminders.instant),{target:{value:'2027-11-07T01:30-04:00'}});
 expect((screen.getByLabelText(en.reminders.instant) as HTMLInputElement).value).toBe('2027-11-07T01:30-04:00');
}
function emit(event: string, id: string|null) {
 for (const f of [...auth.listeners]) f(event,id ? {user:{id},access_token:`jwt-${id}`} : null);
}
it.each(['sign-out/sign-in','A/B/A'])(
 'actual drawer with real Auth subscription clears the visible private reminder after batched %s', async transition => {
 await mount();
 const before=ownAccountWork().epoch;
 const reads=auth.reads;
 await act(async()=>{
  emit(transition==='A/B/A'?'SIGNED_IN':'SIGNED_OUT',transition==='A/B/A'?'real-auth-b':null);
  emit('SIGNED_IN',auth.user.id);
  await query.refetchQueries({queryKey:AUTH_USER_KEY});
 });
 await waitFor(()=>expect(query.isFetching()).toBe(0));
 expect(auth.reads).toBeGreaterThan(reads);
 expect(ownAccountWork().epoch).toBe(before+2);
 // Binding assertion: current UI clears when Auth processing finishes.
 await waitFor(()=>expect((screen.queryByLabelText(en.reminders.instant) as HTMLInputElement|null)?.value ?? '').toBe(''),{timeout:1000});
});
it('actual drawer retains its draft on same-account token refresh',async()=>{
 await mount();const before=ownAccountWork().epoch;
 await act(async()=>{emit('TOKEN_REFRESHED',auth.user.id);await query.refetchQueries({queryKey:AUTH_USER_KEY});});
 expect(ownAccountWork().epoch).toBe(before);
 expect((screen.getByLabelText(en.reminders.instant) as HTMLInputElement).value).toBe('2027-11-07T01:30-04:00');
});
