function holdSession() {
  let resolve!: (value: {
    data: { session: { user: { id: string } } };
  }) => void;
  let reject!: (error: Error) => void;
  const promise = new Promise<{ data: { session: { user: { id: string } } | null }; error?: Error }>(
    (a, b) => {
      resolve = a;
      reject = b;
    },
  );
  auth.reply = () => promise;
  return {
    resolve: (id = "real-auth-a") => {
      auth.user = { id };
      resolve({ data: { session: { user: { id } } } });
    },
    reject,
  };
}
