async function settled() {
  await waitFor(() => expect(query.getQueryData<{id:string}>(AUTH_USER_KEY)?.id).toBe(auth.user.id));
  await waitFor(() => expect(query.isFetching()).toBe(0));
}
function event(name: string, id: string|null) {
  for (const f of [...auth.listeners]) f(name, id ? {user:{id}} : null);
}
it.each(["refresh", "replacement", "sign-out/sign-in", "A/B/A"])(
  "the actual Auth and HelpDrawer preserve or reset the lesson composer on %s", async mode => {
    render(ui(<HelpDrawer open lessonContext={lessonContext} onClose={closeHelp}/>));
    await settled();
    fireEvent.click(screen.getByRole("button", {name:/AI Companion/}));
    const input = await screen.findByLabelText("Message Companion") as HTMLTextAreaElement;
    fireEvent.change(input, {target:{value:"Private lesson draft"}});
    input.focus();
    const before=ownAccountWork();
    await act(async () => {
      if (mode === "refresh") event("TOKEN_REFRESHED", auth.user.id);
      else if (mode === "replacement") {
        auth.user={id:"real-auth-b"};
        event("SIGNED_IN",auth.user.id);
      } else {
        if(mode === "sign-out/sign-in") event("SIGNED_OUT",null);
        else event("SIGNED_IN","real-auth-b");
        event("SIGNED_IN",auth.user.id);
      }
    });
    await settled();
    const after=screen.getByLabelText("Message Companion") as HTMLTextAreaElement;
    if (mode === "refresh") {
      expect(ownAccountWork()).toEqual(before);
      expect(after).toBe(input);
      expect(after.value).toBe("Private lesson draft");
      expect(document.activeElement).toBe(input);
    } else {
      expect(ownAccountWork().epoch).toBe(before.epoch+(mode === "replacement" ? 1 : 2));
      expect(after).not.toBe(input);
      expect(after.value).toBe("");
      expect(screen.queryByDisplayValue("Private lesson draft")).toBeNull();
    }
  }
);
it.each(["companion", "lesson"])(
  "the hydrated actual %s retains a typed draft and focus when pending Auth resolves", async surface => {
    noteAccount("prior-draft-account");noteAccount("real-auth-a");
    let resolveSession!: (value: {data:{session:{user:{id:string}}}})=>void;
    const pending=new Promise<{data:{session:{user:{id:string}}}}>(resolve=>{resolveSession=resolve;});
    auth.reply=()=>pending;
    const node=ui(surface === "companion" ? <CompanionPage/> : <InLessonChat conversationId={null} onConversationCreated={()=>{}} lessonContext={lessonContext}/>);
    const container=document.createElement("div");
    container.innerHTML=renderToString(node);document.body.append(container);
    const input=container.querySelector("textarea")!;
    expect(input).toBeTruthy();input.focus();
    const before=ownAccountWork();
    const onRecoverableError=vi.fn();
    let root: ReturnType<typeof hydrateRoot>|undefined;
    try {
      await act(async()=>{root=hydrateRoot(container,node,{onRecoverableError});});
      expect(container.querySelector("textarea")).toBe(input);
      fireEvent.change(input,{target:{value:"Private hydrated draft"}});
      input.focus();
      expect(query.getQueryState(AUTH_USER_KEY)?.status).toBe("pending");
      await act(async()=>resolveSession({data:{session:{user:{id:"real-auth-a"}}}}));
      await settled();
      expect(ownAccountWork()).toEqual(before);
      expect(container.querySelector("textarea")).toBe(input);
      expect(input.value).toBe("Private hydrated draft");
      expect(document.activeElement).toBe(input);
      expect(onRecoverableError).not.toHaveBeenCalled();
    } finally {
      await act(async()=>root?.unmount());container.remove();
    }
  }
);
